local function guarded(callback)
    local failed = false
    return function(...)
        local ok, err = pcall(callback, ...)
        if not ok and not failed then
            failed = true
            log(255, 80, 80, tostring(err))
        end
    end
end

local function align16(addr)
    return floor((addr + 15) / 16) * 16
end

local function has_bytes(addr, ...)
    for i = 1, select("#", ...) do
        if u8(addr + i - 1) ~= select(i, ...) then
            return false
        end
    end
    return true
end

local function vfunc(instance, index)
    return u32(u32(instance) + index * 4)
end

local function init_entity_filter(base)
    local vtable, should_hit = base + RVA.list_filter_vtable, base + RVA.list_filter_should_hit
    if u32(vtable) ~= should_hit then
        return
    end

    if
        not has_bytes(should_hit + 12, 0x8B, 0x77, 0x1C)
        or not has_bytes(base + RVA.game_rules - 2, 0x8B, 0x0D)
    then
        return
    end

    trace.ignored = ffi.new("uint32_t[?]", MAX_IGNORED)
    trace.entity_filter_data = ffi.new("uint32_t[9]")
    trace.entity_filter_data[0] = vtable
    trace.entity_filter_data[2] = COLLISION_GROUP_PROJECTILE
    trace.entity_filter_data[4] = address(trace.ignored)
    trace.entity_filter_data[5] = MAX_IGNORED
    trace.entity_filter = ffi.cast("void *", trace.entity_filter_data)
    trace.game_rules = u32(base + RVA.game_rules)
end

local function init_trace(base)
    local vtable, should_hit = base + RVA.world_filter_vtable, base + RVA.world_filter_should_hit
    if not has_bytes(base, 0x4D, 0x5A) then
        return
    end
    if u32(vtable) ~= should_hit or not has_bytes(should_hit, 0x32, 0xC0, 0xC2) then
        return
    end

    local interface = client.create_interface("engine.dll", "EngineTraceClient004")
    if not interface then
        return
    end

    local instance = address(interface)
    trace.self = ffi.cast("void *", instance)
    trace.call = ffi.cast("ge_trace_ray_fn", vfunc(instance, VFUNC.trace_ray))

    trace.filter_data = ffi.new("uint32_t[4]")
    trace.filter_data[0] = vtable
    trace.filter = ffi.cast("void *", trace.filter_data)
    init_entity_filter(base)

    trace.ray_data = ffi.new("uint8_t[128]")
    local ray = align16(address(trace.ray_data))
    trace.ray = ffi.cast("float *", ray)
    trace.ray_flags = ffi.cast("uint8_t *", ray)
    trace.ray_ptr = ffi.cast("void *", ray)

    trace.result_data = ffi.new("uint8_t[192]")
    local result = align16(address(trace.result_data))
    trace.result = ffi.cast("float *", result)
    trace.result_flags = ffi.cast("uint8_t *", result)
    trace.result_ptr = ffi.cast("void *", result)

    trace.ready = true
end

local function trace_hull(x, y, z, dx, dy, dz, mask, half, entities)
    local ray = trace.ray
    ray[0], ray[1], ray[2] = x, y, z
    ray[4], ray[5], ray[6] = dx, dy, dz
    ray[8], ray[9], ray[10] = 0, 0, 0
    ray[12], ray[13], ray[14] = half, half, half
    trace.ray_flags[68] = half == 0 and 1 or 0
    trace.ray_flags[69] = (dx ~= 0 or dy ~= 0 or dz ~= 0) and 1 or 0

    ffi.fill(trace.result_data, 192)
    trace.call(
        trace.self,
        trace.ray_ptr,
        mask,
        entities and trace.entity_filter or trace.filter,
        trace.result_ptr
    )

    local result = trace.result
    return result[11],
        result[6],
        result[7],
        result[8],
        result[3],
        result[4],
        result[5],
        trace.result_flags[55] ~= 0
end

local function trace_entity()
    local pointer = ffi.cast("uint32_t *", ffi.cast("uint8_t *", trace.result_ptr) + OFFSET.trace_entity)[0]
    if pointer == 0 then
        return 0, 0
    end
    return i32(pointer + OFFSET.entity_index), pointer
end

local function trace_hit_sky()
    if not trace.enabled then
        return false
    end
    local flags = ffi.cast("uint16_t *", ffi.cast("uint8_t *", trace.result_ptr) + OFFSET.trace_surface)[0]
    return band(flags, SURF_SKY) ~= 0
end

local function reset_ignored()
    if trace.entity_filter_data then
        trace.entity_filter_data[7] = 0
    end
end

local function ignore_entity(pointer)
    local filter = trace.entity_filter_data
    local count = filter[7]
    if count >= MAX_IGNORED then
        return false
    end
    trace.ignored[count] = pointer
    filter[7] = count + 1
    return true
end

local function trace_solid(x, y, z, dx, dy, dz, mask, half)
    if not trace.entities then
        local fraction, nx, ny, nz, ex, ey, ez, start_solid = trace_hull(x, y, z, dx, dy, dz, mask, half)
        return fraction, nx, ny, nz, ex, ey, ez, start_solid, 0, 0
    end

    while true do
        local fraction, nx, ny, nz, ex, ey, ez, start_solid =
            trace_hull(x, y, z, dx, dy, dz, mask, half, true)
        local entindex, pointer = trace_entity()
        local player = entindex >= 1 and entindex <= globals.maxplayers()
        if not player or not ignore_entity(pointer) then
            return fraction, nx, ny, nz, ex, ey, ez, start_solid, entindex, pointer
        end
    end
end

local function sweep_hull(x, y, z, dx, dy, dz)
    local fraction, nx, ny, nz, ex, ey, ez, start_solid, entindex, pointer =
        trace_solid(x, y, z, dx, dy, dz, MASK_GRENADE, GRENADE_SIZE)
    if start_solid then
        fraction, nx, ny, nz, ex, ey, ez, start_solid, entindex, pointer =
            trace_solid(x - dx, y - dy, z - dz, dx * 2, dy * 2, dz * 2, MASK_LINE, 0)
    end
    return fraction, nx, ny, nz, ex, ey, ez, entindex, pointer
end

local HULL_POINTS = { { 0, 0, 0 } }
for _, x in ipairs({ -GRENADE_SIZE, GRENADE_SIZE }) do
    for _, y in ipairs({ -GRENADE_SIZE, GRENADE_SIZE }) do
        for _, z in ipairs({ -GRENADE_SIZE, GRENADE_SIZE }) do
            HULL_POINTS[#HULL_POINTS + 1] = { x, y, z }
        end
    end
end
local PROBE_SIGNS = { { 1, 1 }, { -1, 1 }, { 1, -1 }, { -1, -1 } }

local trace_ignore = 0

local function trace_ray(x, y, z, dx, dy, dz)
    local ex, ey, ez = x + dx, y + dy, z + dz
    local fraction, entindex = client_trace_line(trace_ignore, x, y, z, ex, ey, ez)
    if fraction < 1 and entindex and entindex >= 1 and entindex <= 64 and entindex ~= trace_ignore then
        fraction = client_trace_line(entindex, x, y, z, ex, ey, ez)
    end
    return fraction
end

local function probe_normal(x, y, z, dx, dy, dz, fraction)
    local length = sqrt(dx * dx + dy * dy + dz * dz)
    local ux, uy, uz = dx / length, dy / length, dz / length
    local rx, ry, rz = 0, 0, 1
    if abs(uz) > 0.9 then
        rx, ry, rz = 1, 0, 0
    end

    local ax, ay, az = uy * rz - uz * ry, uz * rx - ux * rz, ux * ry - uy * rx
    local a_length = sqrt(ax * ax + ay * ay + az * az)
    ax, ay, az = ax / a_length, ay / a_length, az / a_length
    local bx, by, bz = uy * az - uz * ay, uz * ax - ux * az, ux * ay - uy * ax

    for _, sign in ipairs(PROBE_SIGNS) do
        local sa, sb = sign[1], sign[2]
        local fa = trace_ray(x + ax * sa, y + ay * sa, z + az * sa, dx, dy, dz)
        local fb = trace_ray(x + bx * sb, y + by * sb, z + bz * sb, dx, dy, dz)
        if fa < 1 and fb < 1 then
            local px, py, pz =
                ax * sa + dx * (fa - fraction), ay * sa + dy * (fa - fraction), az * sa + dz * (fa - fraction)
            local qx, qy, qz =
                bx * sb + dx * (fb - fraction), by * sb + dy * (fb - fraction), bz * sb + dz * (fb - fraction)
            local nx, ny, nz = py * qz - pz * qy, pz * qx - px * qz, px * qy - py * qx
            local n_length = sqrt(nx * nx + ny * ny + nz * nz)
            if n_length > 1e-6 then
                nx, ny, nz = nx / n_length, ny / n_length, nz / n_length
                if nx * ux + ny * uy + nz * uz > 0 then
                    nx, ny, nz = -nx, -ny, -nz
                end
                return nx, ny, nz
            end
        end
    end
    return -ux, -uy, -uz
end

local function sweep_rays(x, y, z, dx, dy, dz)
    local fraction, first = 2, 0
    for i = 1, #HULL_POINTS do
        local point = HULL_POINTS[i]
        local hit = trace_ray(x + point[1], y + point[2], z + point[3], dx, dy, dz)
        if hit < fraction then
            fraction, first = hit, i
        end
    end
    if fraction >= 1 then
        return 1, 0, 0, 0, x + dx, y + dy, z + dz
    end

    local point = HULL_POINTS[first]
    local nx, ny, nz = probe_normal(x + point[1], y + point[2], z + point[3], dx, dy, dz, fraction)
    return fraction, nx, ny, nz, x + dx * fraction, y + dy * fraction, z + dz * fraction
end

local function sweep(x, y, z, dx, dy, dz)
    if trace.enabled then
        return sweep_hull(x, y, z, dx, dy, dz)
    end
    local fraction, nx, ny, nz, ex, ey, ez = sweep_rays(x, y, z, dx, dy, dz)
    return fraction, nx, ny, nz, ex, ey, ez, 0, 0
end

local function find_burn_point(x, y, z)
    if trace.enabled then
        local fraction, _, _, _, _, _, ground = trace_solid(x, y, z + 10, 0, 0, -138, MASK_SOLID, 0)
        if fraction < 1 then
            return x, y, ground + 2
        end
    else
        local fraction = client_trace_line(trace_ignore, x, y, z + 10, x, y, z - 128)
        if fraction < 1 then
            return x, y, z + 10 - 138 * fraction + 2
        end
    end
end

local function trace_is_sane(player)
    local ex, ey, ez = client_eye_position()
    local pitch, yaw = client_camera_angles()
    if not ex or not pitch then
        return false
    end

    local cp, sp, cy, sy = cos(rad(pitch)), sin(rad(pitch)), cos(rad(yaw)), sin(rad(yaw))
    local matches = 0
    for _, direction in ipairs({
        { 0, 0, -1 },
        { 0, 0, 1 },
        { cp * cy, cp * sy, -sp },
        { cy, sy, 0 },
        { -cy, -sy, 0 },
        { sy, -cy, 0 },
    }) do
        local dx, dy, dz = direction[1] * 1500, direction[2] * 1500, direction[3] * 1500
        local expected = client_trace_line(player, ex, ey, ez, ex + dx, ey + dy, ez + dz)
        local fraction, nx, ny, nz = trace_hull(ex, ey, ez, dx, dy, dz, MASK_SOLID, 0)
        if fraction < 1 and abs(sqrt(nx * nx + ny * ny + nz * nz) - 1) > 0.02 then
            return false
        end
        if fraction < 1 and abs(fraction - expected) <= 0.03 then
            matches = matches + 1
        end
    end
    return matches >= 2
end

local function check_trace(player)
    local now = globals_realtime()
    if trace.ready and not trace.enabled and now >= (trace.checked or 0) then
        trace.checked = now + 1
        trace.enabled = trace_is_sane(player)
    end
    trace.entities = trace.enabled and trace.entity_filter ~= nil and u32(trace.game_rules) ~= 0
end

local function lookup_model_name(index)
    if not model_info or index <= 0 then
        return ""
    end
    local model = model_info.get_model(model_info.self, index)
    if model == nil then
        return ""
    end
    local name = model_info.get_name(model_info.self, model)
    return name ~= nil and ffi.string(name):lower() or ""
end

local model_names = {}

local function parse_keyvalues(text)
    local position, length = 1, #text

    local function token()
        while true do
            position = text:find("%S", position) or length + 1
            if text:sub(position, position + 1) ~= "//" then
                break
            end
            position = (text:find("\n", position, true) or length) + 1
        end
        if position > length then
            return
        end

        local first = text:sub(position, position)
        if first == '"' then
            local close = text:find('"', position + 1, true) or length + 1
            local value = text:sub(position + 1, close - 1)
            position = close + 1
            return value, true
        elseif first == "{" or first == "}" then
            position = position + 1
            return first, false
        end

        local stop = text:find('[%s{}"]', position) or length + 1
        local value = text:sub(position, stop - 1)
        position = stop
        return value, true
    end

    local function section()
        local result = {}
        while true do
            local key, key_is_text = token()
            if not key or (key == "}" and not key_is_text) then
                return result
            end
            key = key:lower()

            local value, value_is_text = token()
            if not value then
                return result
            end
            if value == "{" and not value_is_text then
                local blocks = result[key]
                if type(blocks) ~= "table" then
                    blocks = {}
                    result[key] = blocks
                end
                blocks[#blocks + 1] = section()
            elseif result[key] == nil then
                result[key] = value
            end
        end
    end

    return section()
end

local function first_section(tree, key)
    local blocks = tree[key]
    return type(blocks) == "table" and blocks[1] or nil
end

local function to_integer(text)
    local number = tonumber(text) or 0
    return number < 0 and ceil(number) or floor(number)
end

local breakables = {}

local function propdata_base(name)
    local bases = breakables.bases
    if not bases then
        bases = {}
        breakables.bases = bases

        local found, text = pcall(function()
            return readfile(PROPDATA_FILE)
        end)
        local root = found and type(text) == "string" and first_section(parse_keyvalues(text), "propdata.txt")
        if root then
            for key, blocks in pairs(root) do
                if type(blocks) == "table" then
                    bases[key] = blocks[1]
                end
            end
        else
            log(255, 180, 80, "could not read " .. PROPDATA_FILE .. ", breakable props are treated as solid")
        end
    end
    return bases[name:lower()]
end

local function apply_propdata(prop, section, interactions, depth)
    local base = section.base
    base = type(base) == "string" and base ~= "" and depth < 8 and propdata_base(base)
    if base then
        apply_propdata(prop, base, base, depth + 1)
    end

    prop.club = tonumber(section["dmg.club"]) or prop.club
    if section.health then
        prop.health = to_integer(section.health)
    end
    if section.breakable_model then
        prop.gib_model = section.breakable_model
    end
    if section.breakable_count then
        prop.gib_count = to_integer(section.breakable_count)
    end

    for _, interaction in ipairs(INTERACTIONS) do
        local block = first_section(interactions, interaction[1])
        local value = block and block[interaction[2]]
        if type(value) == "string" and value:lower() == interaction[3] then
            prop.interacts = true
            if interaction[2] == "melee_immune" then
                prop.melee_immune = true
            end
        end
    end
end

local function read_prop(model_index)
    local text = model_keyvalues(model_index)
    if not text then
        return
    end

    local root = parse_keyvalues(text)
    root = first_section(root, "mdlkeyvalue") or root
    local section = first_section(root, "prop_data")
    if not section then
        return
    end

    local prop = { health = 0, club = 1, minimum = 0, gib_count = 0 }
    apply_propdata(prop, section, root, 0)

    local collision = collision_keyvalues(model_index)
    local gibs = collision and parse_keyvalues(collision)["break"]
    local gib_count = type(gibs) == "table" and #gibs or 0
    if not collision then
        gib_count = 1
    end
    if gib_count == 0 and prop.gib_model and prop.gib_model ~= "" then
        gib_count = prop.gib_count
    end

    if prop.health ~= 0 and (gib_count > 0 or prop.interacts) then
        return prop
    end
end

local function prop_data(model_index)
    local props = breakables.props
    if not props then
        props = {}
        breakables.props = props
    end

    local prop = props[model_index]
    if prop == nil then
        prop = read_prop(model_index) or false
        props[model_index] = prop
    end
    return prop or nil
end

local function read_brushes()
    local brushes = {}
    for block in (entity_lump() or ""):gmatch("%b{}") do
        local model = block:find('"classname"%s+"func_breakable"') and block:match('"model"%s+"(%*%d+)"')
        if model then
            local function value(key)
                return to_integer(block:match('"' .. key .. '"%s+"([^"]*)"'))
            end
            brushes[model] = {
                health = value("health"),
                material = value("material"),
                flags = value("spawnflags"),
                propdata = value("propdata"),
                minimum = value("minhealthdmg"),
            }
        end
    end
    return brushes
end

local function brush_data(model_index)
    if not breakables.brushes then
        breakables.brushes = read_brushes()
    end

    local brush = { health = 1, club = cvar_float("func_breakdmg_club", 1.5), minimum = 0 }
    local keys = breakables.brushes[model_name(model_index)]
    if not keys then
        return
    end

    brush.health, brush.minimum = keys.health, keys.minimum
    local base = BRUSH_PROPDATA[keys.propdata]
    base = base and propdata_base(base)
    if base then
        apply_propdata(brush, base, base, 0)
    end

    if brush.health == 0 or band(keys.flags, SF_BREAK_TRIGGER_ONLY) ~= 0 then
        brush.immune = true
        if keys.material == MATERIAL_GLASS then
            brush.health = 1
        end
    end
    if keys.material == MATERIAL_UNBREAKABLE_GLASS then
        brush.immune = true
    end
    return brush
end

local function breakable(entindex)
    local kind = CLASSES[entity.get_classname(entindex) or ""]
    if not kind then
        return
    end

    local model_index = entity_get_prop(entindex, "m_nModelIndex") or 0
    local data
    if kind == "glass" then
        data = GLASS
    elseif kind == "brush" then
        data = brush_data(model_index)
        if not data then
            return
        end
    else
        data = prop_data(model_index)
    end

    return {
        kind = kind,
        data = data,
        broken = false,
        health = data and data.health or 0,
        damageable = data ~= nil and not data.immune,
    }
end

local function club(target, damage)
    if target.broken then
        return
    end
    if target.kind == "glass" then
        target.health, target.broken = 0, true
        return
    end

    local data = target.data
    if not target.damageable or damage < data.minimum then
        return
    end
    damage = damage * data.club
    if target.kind ~= "brush" and (damage < data.minimum or data.melee_immune) then
        return
    end

    local health = target.health - damage
    target.health = health < 0 and ceil(health) or floor(health)
    target.broken = target.health <= 0
end

local function is_standable(entindex)
    return band(entity_get_prop(entindex, "m_usSolidFlags") or 0, FSOLID_NOT_STANDABLE) == 0
end

local function simulate(x, y, z, vx, vy, vz, kind, age, path)
    local molotov = kind == "molotov"
    local dt = globals_tickinterval()
    local fuse = molotov and FUSE_MOLOTOV or FUSE_TIMED
    local gravity = GRAVITY_SCALE * cvar.sv_gravity:get_float()
    local speed_limit = cvar.sv_maxvelocity:get_float()
    local think_ticks = max(1, floor(0.5 + THINK_INTERVAL / dt))
    local last_tick = ceil((fuse + 1) / dt)
    local asleep, bounces = false, 0
    local touched = {}
    local next_touch = 0

    reset_ignored()

    local function record(time, collision)
        if path then
            local n = #path
            path[n + 1], path[n + 2], path[n + 3] = x, y, z
            path.times = path.times or {}
            path.times[n / 3 + 1] = time or 0
            if collision then
                path.collisions = path.collisions or {}
                path.collisions[#path.collisions + 1] = { x, y, z, time or 0 }
            end
        end
    end

    local function detonate(time)
        record(time)
        if not molotov then
            return x, y, z, time
        end
        local bx, by, bz = find_burn_point(x, y, z)
        return bx, by, bz, time
    end

    local function breakable_at(entindex)
        if entindex <= 0 then
            return
        end
        local target = touched[entindex]
        if target == nil then
            target = breakable(entindex) or false
            touched[entindex] = target
        end
        return target or nil
    end

    local function ignites(nz, target)
        if not molotov or nz < DETONATE_SLOPE or trace_hit_sky() then
            return false
        end
        return not (target and (target.kind == "glass" or target.kind == "brush"))
    end

    local function ignite(time, entindex)
        if entindex > 0 then
            return detonate(time)
        end
        record(time, true)
        return x, y, z, time
    end

    local function breaks_through(target)
        if target.kind == "physics" and target.broken then
            return false
        end
        if (target.kind == "dynamic" or target.kind == "physics") and not target.damageable then
            return false
        end
        club(target, BREAK_DAMAGE)
        return target.health <= 0
    end

    for tick = 0, last_tick do
        local elapsed = age + tick
        if elapsed % think_ticks == 0 and elapsed * dt > fuse then
            return detonate(tick * dt)
        end
        record(tick * dt)

        if not asleep then
            local falling = vz - gravity * dt
            local mx, my, mz = vx * dt, vy * dt, (vz + falling) / 2 * dt
            vx, vy, vz = clamp(vx, speed_limit), clamp(vy, speed_limit), clamp(falling, speed_limit)

            local fraction, nx, ny, nz, entindex, pointer
            fraction, nx, ny, nz, x, y, z, entindex, pointer = sweep(x, y, z, mx, my, mz)

            if fraction < 1 then
                record((tick + fraction) * dt, true)
                local target = breakable_at(entindex)

                if ignites(nz, target) then
                    return ignite((tick + fraction) * dt, entindex)
                end
                if
                    kind == "flash"
                    and target
                    and not target.broken
                    and (target.kind ~= "brush" or target.damageable)
                    and tick * dt >= next_touch
                    and vx * vx + vy * vy + vz * vz > TOUCH_SPEED * TOUCH_SPEED
                then
                    club(target, TOUCH_DAMAGE)
                    next_touch = tick * dt + TOUCH_INTERVAL
                end

                local through = target and breaks_through(target)
                if target and target.broken then
                    ignore_entity(pointer)
                end

                if through then
                    vx, vy, vz = vx * BREAK_SLOWDOWN, vy * BREAK_SLOWDOWN, vz * BREAK_SLOWDOWN
                else
                    local backoff = (vx * nx + vy * ny + vz * nz) * 2
                    vx, vy, vz = vx - nx * backoff, vy - ny * backoff, vz - nz * backoff
                    if abs(vx) < STOP_EPSILON then
                        vx = 0
                    end
                    if abs(vy) < STOP_EPSILON then
                        vy = 0
                    end
                    if abs(vz) < STOP_EPSILON then
                        vz = 0
                    end
                    vx, vy, vz = vx * ELASTICITY, vy * ELASTICITY, vz * ELASTICITY

                    local speed_sqr = vx * vx + vy * vy + vz * vz
                    local floor_like = nz > 0.7 or (nz > 0.1 and speed_sqr < SLEEP_SPEED_SQR)
                    if floor_like and (entindex <= 0 or is_standable(entindex)) then
                        if speed_sqr < SLEEP_SPEED_SQR then
                            vx, vy, vz, asleep = 0, 0, 0, true
                        elseif speed_sqr > 96000 then
                            local along = (vx * nx + vy * ny + vz * nz) / sqrt(speed_sqr)
                            if along > 0.5 then
                                local keep = (1 - along) + 0.5
                                vx, vy, vz = vx * keep, vy * keep, vz * keep
                            end
                        end
                    end

                    if not asleep then
                        local rest = (1 - fraction) * dt
                        fraction, nx, ny, nz, x, y, z, entindex =
                            sweep(x, y, z, vx * rest, vy * rest, vz * rest)
                        if fraction < 1 then
                            record((tick + 1) * dt, true)
                        end
                        if fraction < 1 and ignites(nz, breakable_at(entindex)) then
                            return ignite((tick + 1) * dt, entindex)
                        end
                    end

                    bounces = bounces + 1
                    if bounces > MAX_BOUNCES then
                        return detonate(tick * dt)
                    end
                end
            end
        end
    end

    return detonate(last_tick * dt)
end

                    local values = saved[name] or {}
                    values[1], values[2], values[3] = entity_get_prop(index, name)
                    saved[name] = values
                end
                target_snapshots[index] = saved
            end
            if saved.alive then
                damage_targets[#damage_targets + 1] = index
            end
        end
        for index in pairs(target_snapshots) do
            if not candidates[index] then
                target_snapshots[index] = nil
            end
        end
    end
    local function after_armor(player, damage, scale)
        local heavy = damage_prop(player, "m_bHasHeavyArmor") == 1
        damage = damage * scale
        if heavy then
            damage = damage * 0.85
        end
        local armor = damage_prop(player, "m_ArmorValue") or 0
        if armor <= 0 or damage <= 0 then
            return damage
        end
        local ratio, bonus, heavy_bonus = 0.5 * HE_ARMOR_RATIO, 0.5, 1
        if heavy then
            ratio, bonus, heavy_bonus = ratio * 0.5, 0.33, 0.33
        end
        local to_health = damage * ratio
        local to_armor = (damage - to_health) * (bonus * heavy_bonus)
        if to_armor > armor then
            to_health = damage - armor / bonus
        end
        return to_health
    end

    local function blast_damage(player, local_player, x, y, z, damage, radius, scale)
        local ox, oy, oz = damage_prop(player, "m_vecOrigin")
        if not ox then
            return 0, after_armor(player, damage, scale)
        end
        local ex, ey, ez
        if player == local_player then
            ex, ey, ez = client_eye_position()
        end
        if not ex then
            local view_z = damage_prop(player, "m_vecViewOffset[2]") or 64
            ex, ey, ez = ox, oy, oz + view_z
        end
        z = z + 1
        local point_blank = after_armor(player, damage, scale)
        local dx, dy, dz = ex - x, ey - y, ez - z
        local sigma = radius / 3
        local falloff = exp(-(dx * dx + dy * dy + dz * dz) / (2 * sigma * sigma))
        if after_armor(player, damage * falloff, scale) < 0.5 then
            return 0, point_blank
        end
        local rx, ry = PLAYER_HALF_WIDTH, 0
        local visible = 0.40 * exposed(x, y, z, ex, ey, ez)
            + 0.20 * exposed(x, y, z, ox, oy, oz + PLAYER_HEIGHT)
            + 0.20 * exposed(x, y, z, ox, oy, oz)
            + 0.10 * exposed(x, y, z, ox - rx, oy - ry, oz + 36)
            + 0.10 * exposed(x, y, z, ox + rx, oy + ry, oz + 36)
        if visible <= 0 then
            return 0, point_blank
        end
        return max(0, floor(after_armor(player, damage * falloff * visible, scale) + 0.5)), point_blank
    end

    local function grenade_kind(entindex, radius)
        local model = model_name(entity_get_prop(entindex, "m_nModelIndex") or 0)
        if model:find("frag", 1, true) then
            return "he"
        end
        if model:find("flash", 1, true) then
            return "flash"
        end
        if radius > 0 and radius <= HE_MIN_RADIUS then
            return "flash"
        end
        return "he"
    end
    local function entity_from_handle(handle)
        if handle and handle > 0 then
            return band(handle, 0x7FF)
        end
    end
    local function is_threat(kind, own, friendly, friendly_fire, self_scale)
        if kind == "flash" then
            return not own
        end

        return true
    end
    local tracked, fire_states, retired = {}, {}, {}
    local render_clock
    local damage_error_logged = false
    local function reset()
        tracked, fire_states, retired, target_snapshots, damage_targets = {}, {}, {}, {}, {}
        render_clock = nil
        fallback_projectiles, fallback_scan_at, targets_updated = {}, 0, -1
        targets_count, targets_revision = -1, 0
    end
    local function finish_projectile(index)
        retired[index] = retired[index] or globals_realtime()
        local state = tracked[index]
        if state then
            state.path = {}
            state.finished = true
            state.finished_at = state.finished_at or globals_realtime()
            if state.last_warning then
                state.last_warning.active = false
            end
        end
    end

    local function collect_damage(
        grenade,
        local_player,
        thrower,
        own,
        friendly,
        self_scale,
        team_scale,
        raw_damage,
        radius
    )
        local entries, local_damage, max_damage = {}, 0, 0
        local targets = damage_targets
        for _, target in ipairs(targets) do
            if
                entity.is_alive(target)
                or (dormant(target) and target_snapshots[target] and target_snapshots[target].alive)
            then
                local thrower_team = thrower and damage_prop(thrower, "m_iTeamNum")
                local target_team = damage_prop(target, "m_iTeamNum")
                local same_team = thrower_team ~= nil and thrower_team == target_team
                local scale = target == thrower and self_scale or (same_team and team_scale or 1)
                if scale > 0 then
                    local damage, maximum = blast_damage(
                        target,
                        local_player,
                        grenade.x,
                        grenade.y,
                        grenade.z,
                        raw_damage,
                        radius,
                        scale
                    )
                    max_damage = target == local_player and maximum or max_damage
                    if damage > 0 then
                        local local_hit = target == local_player
                        entries[#entries + 1] = {
                            damage = damage,
                            local_hit = local_hit,
                            index = target,
                            enemy = target ~= local_player
                                and target_snapshots[target]
                                and target_snapshots[target].enemy == true,
                        }
                        if local_hit then
                            local_damage = damage
                        end
                    end
                end
            end
        end
        table.sort(entries, function(a, b)
            if a.local_hit ~= b.local_hit then
                return a.local_hit
            end
            return a.damage > b.damage
        end)
        return entries, local_damage, max_damage
    end

    local function add_projectiles(warnings, player, friendly_fire)
        local now, dt = globals_realtime(), globals_tickinterval()
        local self_scale = cvar_float("ff_damage_reduction_grenade_self", 1)
        local team_scale = friendly_fire and cvar_float("ff_damage_reduction_grenade", 0.25) or 0
        local half_latency = (client.latency() or 0) / 2
        local seen = {}
        local candidates = {}
        for _, projectile in ipairs(PROJECTILES) do
            local ok, entities = pcall(entity_get_all, projectile[1])
            for _, index in ipairs(ok and entities or {}) do
                candidates[index] = projectile[2] or candidates[index] or false
            end
        end

        if now >= fallback_scan_at then
            fallback_scan_at = now + 0.25
            fallback_projectiles = {}
            local ok_all, all = pcall(entity_get_all)
            for _, index in ipairs(ok_all and all or {}) do
                if candidates[index] == nil and entity.get_classname then
                    local class = entity.get_classname(index) or ""
                    if class:lower():find("projectile", 1, true) then
                        local model = entity_get_prop(index, "m_nModelIndex") or 0
                        local name = model_name(model)
                        local kind
                        if name:find("frag", 1, true) or name:find("hegrenade", 1, true) then
                            kind = "he"
                        elseif name:find("molotov", 1, true) or name:find("incendiary", 1, true) then
                            kind = "molotov"
                        end
                        if kind then
                            fallback_projectiles[index] = { kind = kind, model = model }
                        end
                    end
                end
            end
        end
        for index, cached in pairs(fallback_projectiles) do
            if candidates[index] == nil then
                if entity_get_prop(index, "m_nModelIndex") == cached.model then
                    candidates[index] = cached.kind
                else
                    fallback_projectiles[index] = nil
                end
            end
        end
        for entindex, forced_kind in pairs(candidates) do
            local molotov = forced_kind == "molotov"
            do
                local x, y, z = entity_get_prop(entindex, "m_vecOrigin")

                local exploded = (entity_get_prop(entindex, "m_nExplodeEffectTickBegin") or 0) > 0
                local thrower = entity_from_handle(entity_get_prop(entindex, "m_hThrower"))
                local own = thrower == player
                local friendly = not own and thrower ~= nil and entity_is_enemy(thrower) == false
                local radius = entity_get_prop(entindex, "m_DmgRadius") or 0
                local kind = forced_kind or (molotov and "molotov" or grenade_kind(entindex, radius))
                local previous = tracked[entindex]
                if
                    exploded or (previous and previous.end_time and globals.curtime() >= previous.end_time)
                then
                    finish_projectile(entindex)
                end
                if
                    not seen[entindex]
                    and not retired[entindex]
                    and x
                    and not exploded
                    and is_threat(kind, own, friendly, friendly_fire, self_scale)
                then
                    seen[entindex] = true
                    local grenade = tracked[entindex]
                    if not grenade then
                        grenade = {
                            first_seen = now,
                            initial_age = half_latency + dt,
                            simulated = -1,
                            alpha = 0,
                            last_frame = now,
                        }
                        tracked[entindex] = grenade
                    end
                    grenade.current_position = { x, y, z }
                    local age = grenade.initial_age + now - grenade.first_seen
                    local simtime = entity_get_prop(entindex, "m_flSimulationTime") or globals.curtime()
                    local spawned = entity_get_prop(entindex, "m_flSpawnTime")
                    local vx, vy, vz = entity_get_prop(entindex, "m_vecVelocity")
                    if spawned and spawned > 0 and spawned <= simtime and simtime - spawned < 10 then
                        age = simtime - spawned
                    elseif (entity_get_prop(entindex, "m_nBounces") or 0) == 0 then
                        local ix, iy, iz = entity_get_prop(entindex, "m_vInitialVelocity")
                        local gravity = cvar_float("sv_gravity", 800) * GRAVITY_SCALE
                        if
                            iz
                            and vz
                            and gravity > 0
                            and abs((ix or 0) - (vx or 0)) + abs((iy or 0) - (vy or 0)) < 12
                        then
                            local ballistic_age = (iz - vz) / gravity
                            if ballistic_age >= 0 and ballistic_age < 4 then
                                age = ballistic_age
                            end
                        end
                    end

                    local outdated = grenade.simulated < 0 or grenade.sample_time ~= simtime
                    if outdated then
                        local age_ticks = max(0, floor(age / dt + 0.5))
                        local path = {}
                        local px, py, pz, time_left =
                            simulate(x, y, z, vx or 0, vy or 0, vz or 0, kind, age_ticks, path)
                        grenade.simulated, grenade.path = now, path
                        grenade.sample_time = simtime
                        grenade.path_start = globals.curtime() - max(0, globals.curtime() - simtime)
                        grenade.x, grenade.y, grenade.z = px, py, pz
                        grenade.fuse = px and age_ticks * dt + time_left
                        local predicted_end = px and grenade.path_start + time_left

                        if
                            predicted_end
                            and grenade.end_time
                            and (kind == "he" or grenade.end_time - globals.curtime() <= 0.12)
                        then
                            grenade.end_time = min(grenade.end_time, predicted_end)
                        else
                            grenade.end_time = predicted_end
                        end
                    end
                    if
                        grenade.x
                        and kind == "he"
                        and (
                            now - (grenade.damage_time or -1) >= DAMAGE_INTERVAL
                            or grenade.damage_revision ~= targets_revision
                        )
                    then
                        local raw_damage = entity_get_prop(entindex, "m_flDamage") or 0
                        grenade.damage_time = now
                        grenade.damage_revision = targets_revision
                        local ok, entries, local_damage, maximum = pcall(
                            collect_damage,
                            grenade,
                            player,
                            thrower,
                            own,
                            friendly,
                            self_scale,
                            team_scale,
                            raw_damage >= 1 and raw_damage or HE_DAMAGE,
                            radius >= HE_MIN_RADIUS and radius or HE_RADIUS
                        )
                        if ok then
                            grenade.damage_entries, grenade.damage, grenade.max_damage =
                                entries, local_damage, maximum
                        else
                            grenade.damage_entries, grenade.damage, grenade.max_damage = {}, 0, HE_DAMAGE
                            if not damage_error_logged then
                                damage_error_logged = true
                                log(255, 100, 100, "HE damage labels disabled: " .. tostring(entries))
                            end
                        end
                    end
                    if grenade.x then
                        if kind == "he" then
                            grenade.enemy_danger = false
                            for _, entry in ipairs(grenade.damage_entries or {}) do
                                if entry.enemy and entry.damage > 0 then
                                    grenade.enemy_danger = true
                                    break
                                end
                            end
                        elseif kind == "molotov" and now - (grenade.danger_time or -1) >= DAMAGE_INTERVAL then
                            grenade.danger_time = now
                            grenade.enemy_danger = false
                            for _, enemy in ipairs(damage_targets) do
                                local ox, oy, oz = damage_prop(enemy, "m_vecOrigin")
                                local owner_team = thrower and damage_prop(thrower, "m_iTeamNum")
                                local enemy_team = damage_prop(enemy, "m_iTeamNum")
                                local can_hurt = not thrower
                                    or enemy == thrower
                                    or owner_team ~= enemy_team
                                    or friendly_fire
                                if
                                    target_snapshots[enemy]
                                    and target_snapshots[enemy].enemy
                                    and can_hurt
                                    and ox
                                then
                                    local dx, dy = ox - grenade.x, oy - grenade.y
                                    if
                                        dx * dx + dy * dy > 0
                                        and dx * dx + dy * dy + (oz - grenade.z) ^ 2 < (3 * 39.37) ^ 2
                                        and abs(oz - grenade.z) < 70
                                        and exposed(grenade.x, grenade.y, grenade.z + 4, ox, oy, oz + 36)
                                            > 0
                                    then
                                        grenade.enemy_danger = true
                                        break
                                    end
                                end
                            end
                        end
                    end

                    if grenade.x then
                        grenade.last_seen = now
                        local warning = {
                            x = grenade.x,
                            y = grenade.y,
                            z = grenade.z,
                            remaining = max(0, (grenade.end_time or globals.curtime()) - globals.curtime()),
                            kind = kind,
                            own = own,
                            damage = grenade.damage or 0,
                            max_damage = grenade.max_damage,
                            damage_entries = grenade.damage_entries or {},
                            state = grenade,
                            active = true,
                        }
                        grenade.last_warning = warning
                        warnings[#warnings + 1] = warning
                    end
                end
            end
        end
        for index, finished_at in pairs(retired) do
            if candidates[index] == nil and now - finished_at > 1 then
                retired[index] = nil
            end
        end
        for entindex, grenade in pairs(tracked) do
            if not seen[entindex] then
                local elapsed = now - (grenade.finished_at or grenade.last_seen or now)
                if grenade.last_warning and elapsed < FADE_TIME then
                    local warning = grenade.last_warning
                    warning.active, warning.remaining = false, 0
                    warnings[#warnings + 1] = warning
                else
                    tracked[entindex] = nil
                end
            end
        end
    end

    local function nearest_fire_segment(index, state, x, y, z, px, py, pz, now)
        local raw_count = entity_get_prop(index, "m_fireCount")
        local count = min(64, max(0, raw_count or 1))
        if
            not state.cells
            or state.cell_count ~= count
            or now - (state.cells_time or -1) >= DAMAGE_INTERVAL
        then
            local cells = state.cells or {}
            local used = 0
            for slot = 0, count - 1 do
                local burning = entity_get_prop(index, "m_bFireIsBurning", slot)
                if burning ~= false and burning ~= 0 then
                    local dx = entity_get_prop(index, "m_fireXDelta", slot)
                    local dy = entity_get_prop(index, "m_fireYDelta", slot)
                    local dz = entity_get_prop(index, "m_fireZDelta", slot)
                    if dx and dy and dz then
                        used = used + 1
                        local cell = cells[used] or {}
                        cell[1], cell[2], cell[3] = x + dx, y + dy, z + dz
                        cells[used] = cell
                    elseif not raw_count then
                        used = used + 1
                        cells[used] = { x, y, z }
                    end
                end
            end
            for slot = #cells, used + 1, -1 do
                cells[slot] = nil
            end
            state.cells, state.cells_time, state.cell_count = cells, now, count
        end
        local closest, best_distance, touching
        for _, cell in ipairs(state.cells) do
            local dx, dy, dz = px - cell[1], py - cell[2], pz - cell[3]
            local horizontal = sqrt(dx * dx + dy * dy)
            local edge = max(0, horizontal - 60)
            local distance = sqrt(edge * edge + dz * dz)
            if not best_distance or distance < best_distance then
                best_distance = distance
                local fraction = horizontal > 60 and 60 / horizontal or 0
                closest = { cell[1] + dx * fraction, cell[2] + dy * fraction, cell[3] }
                touching = horizontal <= 60 and abs(dz) < 70
            end
        end
        return closest, best_distance, touching
    end

    local function add_fires(warnings, player, friendly_fire)
        local dt, lifetime, seen = globals_tickinterval(), cvar_float("inferno_flame_lifetime", 7), {}
        local now = globals_realtime()
        local px, py, pz = damage_prop(player, "m_vecOrigin")
        if not px then
            return
        end
        for _, entindex in ipairs(entity_get_all("CInferno")) do
            local x, y, z = entity_get_prop(entindex, "m_vecOrigin")
            local ignited = entity_get_prop(entindex, "m_nFireEffectTickBegin")
            if x and ignited and ignited > 0 then
                local owner = entity_from_handle(entity_get_prop(entindex, "m_hOwnerEntity"))
                local friendly = owner ~= nil and owner ~= player and entity_is_enemy(owner) == false
                local burning = max(0, (globals.tickcount() - ignited) * dt)
                if (not friendly or friendly_fire) and burning < lifetime then
                    local state = fire_states[entindex] or { alpha = 0, last_frame = now, first_seen = now }
                    fire_states[entindex] = state
                    local position, distance, touching =
                        nearest_fire_segment(entindex, state, x, y, z, px, py, pz, now)
                    if position then
                        seen[entindex] = true
                        local frame_time = now - (state.position_time or now)
                        state.position_time = now
                        if state.world_x then
                            state.world_x = approach(state.world_x, position[1], 14, frame_time)
                            state.world_y = approach(state.world_y, position[2], 14, frame_time)
                            state.world_z = approach(state.world_z, position[3], 14, frame_time)
                        else
                            state.world_x, state.world_y, state.world_z =
                                position[1], position[2], position[3]
                        end
                        state.last_seen = now
                        local warning = {
                            x = state.world_x,
                            y = state.world_y,
                            z = state.world_z,
                            fire_distance = distance,
                            touching = touching,
                            remaining = lifetime - burning,
                            kind = "fire",
                            damage = 0,
                            damage_entries = {},
                            state = state,
                            active = true,
                        }
                        state.last_warning = warning
                        warnings[#warnings + 1] = warning
                    end
                end
            end
        end
        for entindex, state in pairs(fire_states) do
            if not seen[entindex] then
                if state.last_warning and now - (state.last_seen or now) < FADE_TIME then
                    local warning = state.last_warning
                    warning.active, warning.remaining = false, 0
                    warnings[#warnings + 1] = warning
                else
                    fire_states[entindex] = nil
                end
            end
        end
    end

    local function visual_time()
        local tick_time, now = globals.curtime(), globals_realtime()
        if not render_clock then
            render_clock = { tick = tick_time, updated = now, value = tick_time }
        elseif render_clock.tick ~= tick_time then
            render_clock.tick, render_clock.updated = tick_time, now
        end
        local time = tick_time + min(max(0, now - render_clock.updated), globals_tickinterval() * 2)
        render_clock.value = max(render_clock.value, time)
        return render_clock.value
    end

    local function store_path_point(points, index, x, y, z, time)
        local point = points[index] or {}
        point[1], point[2], point[3], point[4] = x, y, z, time
        points[index] = point
    end

    local function smooth_path(state)
        local path, points = state.path or {}, state.render_points or {}
        state.render_points = points
        local count = 0
        local elapsed = max(0, visual_time() - (state.path_start or globals.curtime()))
        local times = path.times or {}
        for index = 1, #path / 3 do
            local time = times[index] or (index - 1) * globals_tickinterval()
            if time >= elapsed then
                local offset = (index - 1) * 3
                if count == 0 and index > 1 then
                    local previous_time = times[index - 1] or time - globals_tickinterval()
                    local fraction = saturate((elapsed - previous_time) / max(time - previous_time, 0.0001))
                    count = 1
                    store_path_point(
                        points,
                        count,
                        path[offset - 2] + (path[offset + 1] - path[offset - 2]) * fraction,
                        path[offset - 1] + (path[offset + 2] - path[offset - 1]) * fraction,
                        path[offset] + (path[offset + 3] - path[offset]) * fraction,
                        elapsed
                    )
                end
                count = count + 1
                store_path_point(points, count, path[offset + 1], path[offset + 2], path[offset + 3], time)
            end
        end
        return points, elapsed, count
    end

    local function draw_trajectory(warning, r, g, b, alpha)
        if not ui_get(show_path) or warning.kind == "fire" then
            return
        end
        local points, elapsed, count = smooth_path(warning.state)
        local phase = globals_realtime() * 1.8
        local previous_x, previous_y, previous_time
        for index = 1, count do
            local point = points[index]
            local x, y = renderer.world_to_screen(point[1], point[2], point[3])
            if x and previous_x and (x - previous_x) ^ 2 + (y - previous_y) ^ 2 >= 0.0625 then
                local dx, dy = x - previous_x, y - previous_y
                local length = sqrt(dx * dx + dy * dy)
                local parts = previous_time - elapsed < 0.08 and max(1, min(16, ceil(length / 6))) or 1
                for part = 1, parts do
                    local t0, t1 = (part - 1) / parts, part / parts
                    local time = previous_time + (point[4] - previous_time) * ((t0 + t1) * 0.5)
                    local fade = smoothstep((time - elapsed) / 0.08)
                    local progress = (warning.state.path_start or 0) + time
                    local wave = 0.55 + 0.45 * (0.5 + 0.5 * sin((progress * 3 - phase) * math.pi * 2))
                    local a = floor(alpha * wave * fade)
                    if a > 0 then
                        local ax, ay = previous_x + dx * t0, previous_y + dy * t0
                        local bx, by = previous_x + dx * t1, previous_y + dy * t1
                        renderer.line(ax + 1, ay + 1, bx + 1, by + 1, 0, 0, 0, floor(a * 0.72))
                        renderer.line(ax, ay, bx, by, r, g, b, a)
                    end
                end
            end
            previous_x, previous_y, previous_time = x, y, point[4]
        end

        for _, hit in ipairs((warning.state.path or {}).collisions or {}) do
            if hit[4] >= elapsed then
                local x, y = renderer.world_to_screen(hit[1], hit[2], hit[3])
                if x then
                    local hit_alpha = floor(alpha * smoothstep((hit[4] - elapsed) / 0.08))
                    if hit_alpha > 0 then
                        collision_marker(x, y, 255, 255, 255, hit_alpha)
                    end
                end
            end
        end

        local x, y = renderer.world_to_screen(warning.x, warning.y, warning.z)
        if x then
            local tint = warning.state.enemy_danger and GREEN or RED
            collision_marker(x, y, tint[1], tint[2], tint[3], alpha)
        end
    end

    local function draw_damage_labels(warning)
        if not ui_get(show_damage) or warning.kind == "fire" then
            return
        end
        local x, y = renderer.world_to_screen(warning.x, warning.y, warning.z + 4)
        if not x then
            return
        end
        local labels = {}
        if warning.kind == "he" then
            for _, entry in ipairs(warning.damage_entries) do
                if (entry.local_hit or entry.enemy) and (entry.damage or 0) > 0 and #labels < 4 then
                    labels[#labels + 1] = entry
                end
            end
        elseif warning.kind == "molotov" then
            local local_player = entity.get_local_player()
            for _, target in ipairs(damage_targets) do
                local snapshot = target_snapshots[target]
                if target == local_player or (snapshot and snapshot.enemy) then
                    local ox, oy, oz = damage_prop(target, "m_vecOrigin")
                    if ox then
                        local dx, dy, dz = warning.x - ox, warning.y - oy, warning.z - oz
                        local distance = sqrt(dx * dx + dy * dy + dz * dz)
                        local meters = floor(distance / 39.37 * 10 + 0.5) / 10
                        if distance / 39.37 < 4 and meters > 0 then
                            labels[#labels + 1] = {
                                index = target,
                                text = string.format("%.1f m", meters),
                                distance = distance,
                                local_player = target == local_player,
                                local_hit = target == local_player
                                    and distance <= cvar_float("inferno_max_range", 150),
                            }
                        end
                    end
                end
            end
            table.sort(labels, function(a, b)
                if a.local_player ~= b.local_player then
                    return a.local_player
                end
                if a.distance == b.distance then
                    return a.index < b.index
                end
                return a.distance < b.distance
            end)
            while #labels > 4 do
                table.remove(labels)
            end
        end
        if warning.active == false then
            labels = {}
        end
        local state, now = warning.state, globals_realtime()
        local elapsed = min(0.1, now - (state.labels_frame or now))
        state.labels_frame = now
        state.label_rows = state.label_rows or {}
        local rows, wanted = state.label_rows, {}
        for index, entry in ipairs(labels) do
            local key = entry.index and ("player:" .. entry.index) or "distance"
            wanted[key] = true
            local row = rows[key]
            if not row then
                row = { alpha = 0, offset = 0, key = key }
                rows[key] = row
            end
            row.entry, row.order = entry, index
        end
        local visible, total = {}, 0
        for key, row in pairs(rows) do
            row.alpha = approach(row.alpha, wanted[key] and 1 or 0, wanted[key] and 16 or 20, elapsed)
            if not wanted[key] and row.alpha < 0.008 then
                rows[key] = nil
            else
                local text = row.entry.text or string.format("%d dmg", row.entry.damage)
                if row.text ~= text then
                    row.text = text
                    row.width, row.height = renderer.measure_text("", text)
                end
                total = total + (row.height + 1) * row.alpha
                visible[#visible + 1] = row
            end
        end
        table.sort(visible, function(a, b)
            if a.order == b.order then
                return a.key < b.key
            end
            return a.order < b.order
        end)
        local cursor = 0
        for _, row in ipairs(visible) do
            local entry = row.entry
            local text = row.text
            local tw, th = row.width, row.height

            local target_offset = -6 - total + cursor
            row.offset = approach(row.offset, target_offset, 18, elapsed)
            cursor = cursor + (th + 1) * row.alpha
            local tx, ty = floor(x - tw / 2), floor(y + row.offset - (1 - row.alpha) * 5)
            local text_alpha = row.alpha * (1 - (state.warning_visibility or 0))
            local opacity = floor(255 * text_alpha + 0.5)
            if opacity > 0 then
                local pad = 4
                local half = ceil((tw + pad * 2 + 16) / 2)
                renderer.gradient(
                    x - half,
                    ty - 1,
                    half,
                    th + 2,
                    0,
                    0,
                    0,
                    0,
                    0,
                    0,
                    0,
                    floor(105 * text_alpha),
                    true
                )
                renderer.gradient(x, ty - 1, half, th + 2, 0, 0, 0, floor(105 * text_alpha), 0, 0, 0, 0, true)
                local tr, tg, tb = 255, 255, 255
                if entry.local_hit then
                    tr, tg, tb = 255, 58, 58
                elseif
                    (entry.enemy and (entry.damage or 0) > 50)
                    or (entry.distance and not entry.local_player and entry.distance < 3 * 39.37)
                then
                    tr, tg, tb = GREEN[1], GREEN[2], GREEN[3]
                end
                if glow_texture then
                    renderer.texture(
                        glow_texture,
                        tx - 8,
                        ty - 4,
                        tw + 16,
                        th + 8,
                        0,
                        0,
                        0,
                        floor(90 * text_alpha),
                        "f"
                    )
                end
                renderer_text(tx, ty, tr, tg, tb, opacity, "", 0, text)
            end
        end
    end

    local function warning_risk(warning, fire_reach)
        if warning.kind == "he" then
            if warning.damage > 8 then
                return 1, true
            end
            return saturate(warning.damage / 8), false
        elseif warning.kind == "fire" then
            return 1 - saturate((warning.fire_distance or warning.distance) / 200), warning.touching == true
        elseif warning.kind == "molotov" then
            local risk = 1 - saturate((warning.distance - fire_reach) / max(RANGE - fire_reach, 1))
            return risk, warning.distance <= fire_reach
        end
        return 0, false
    end

    local function screen_position(wx, wy, wz, width, height, margin)
        local half_w, half_h = width / 2 - margin, height / 2 - margin
        local screen_x, screen_y = renderer.world_to_screen(wx, wy, wz)
        if screen_x then
            local dx, dy = screen_x - width / 2, screen_y - height / 2
            local reach = sqrt((dx / half_w) ^ 2 + (dy / half_h) ^ 2)
            local outside = reach > 1
            if outside then
                dx, dy = dx / reach, dy / reach
            end
            return width / 2 + dx, height / 2 + dy, outside
        end
        local ex, ey, ez = client_eye_position()
        local pitch, yaw = client_camera_angles()
        if not ex or not pitch then
            return width / 2, height - margin, true
        end
        local cp, sp, cy, sy = cos(rad(pitch)), sin(rad(pitch)), cos(rad(yaw)), sin(rad(yaw))
        local dx, dy, dz = wx - ex, wy - ey, wz - ez
        local right = dx * sy - dy * cy
        local down = -(dx * (sp * cy) + dy * (sp * sy) + dz * cp)
        local length = sqrt(right * right + down * down)
        if length < 1e-4 then
            right, down, length = 0, 1, 1
        end
        right, down = right / length, down / length
        local scale = 1 / sqrt((right / half_w) ^ 2 + (down / half_h) ^ 2)
        return width / 2 + right * scale, height / 2 + down * scale, true
    end

    local function draw_arrow(x, y, width, height, warning_radius, r, g, b, alpha)
        local dx, dy = x - width / 2, y - height / 2
        local length = max(1, sqrt(dx * dx + dy * dy))
        dx, dy = dx / length, dy / length
        local side_x, side_y = -dy, dx
        local arrow_x, arrow_y = x + dx * (warning_radius + 10), y + dy * (warning_radius + 10)
        local tip_x, tip_y = arrow_x + dx * 5, arrow_y + dy * 5
        local base_x, base_y = arrow_x - dx * 3, arrow_y - dy * 3
        local half_base = 3 * sqrt(3)
        renderer.triangle(
            tip_x + 1,
            tip_y + 1,
            base_x + side_x * half_base + 1,
            base_y + side_y * half_base + 1,
            base_x - side_x * half_base + 1,
            base_y - side_y * half_base + 1,
            0,
            0,
            0,
            floor(alpha * 0.75)
        )
        renderer.triangle(
            tip_x,
            tip_y,
            base_x + side_x * half_base,
            base_y + side_y * half_base,
            base_x - side_x * half_base,
            base_y - side_y * half_base,
            r,
            g,
            b,
            alpha
        )
    end

    local function draw_warning(warning, width, height, fire_reach)
        local state, now = warning.state, globals_realtime()
        local dt = min(0.1, now - (state.last_frame or now))
        state.last_frame = now
        local active = warning.active ~= false
        state.visible_since = state.visible_since or state.first_seen or now
        if active then
            state.fade_started = nil
            local appearing = ease_out_cubic((now - state.visible_since) / 0.24)
            state.alpha, state.scale = appearing, 0.72 + 0.28 * appearing
        else
            if not state.fade_started then
                state.fade_started, state.fade_alpha = now, state.alpha or 0
            end
            local fading = smoothstep((now - state.fade_started) / FADE_TIME)
            state.alpha = state.fade_alpha * (1 - fading)
            state.scale = 1 - 0.18 * fading
        end
        if state.alpha < 0.005 then
            return
        end
        local risk, critical = warning_risk(warning, fire_reach)
        state.risk = approach(state.risk or 0, risk, 8, dt)
        local r, g, b = mix_color(critical and 1 or state.risk)
        local period = critical and 0.28 or mix(105, 38, state.risk) / 100
        local pulse = 0.5 + 0.5 * sin(now * math.pi * 2 / period)
        pulse = smoothstep(pulse)
        local display_alpha = state.alpha * (state.warning_visibility or 0)
        local alpha = floor(255 * display_alpha)
        local radius = (25 + 6 * state.risk + 3 * pulse * state.risk) * state.scale
        local glow_radius = (44 + 2 * pulse * state.risk) * state.scale

        local x, y, offscreen = screen_position(warning.x, warning.y, warning.z + 12, width, height, 90)
        local glow_r, glow_g, glow_b = r, g, b
        local glow_alpha = floor((150 + 55 * state.risk + 35 * pulse * state.risk) * display_alpha)
        if glow_texture then
            renderer.texture(
                glow_texture,
                floor(x - glow_radius),
                floor(y - glow_radius),
                floor(glow_radius * 2),
                floor(glow_radius * 2),
                glow_r,
                glow_g,
                glow_b,
                glow_alpha,
                "f"
            )
        else
            renderer_circle(x, y, glow_r, glow_g, glow_b, floor(glow_alpha * 0.55), radius, 0, 1)
        end

        local icon_offset = warning.kind == "he" and -12 or (warning.kind == "flash" and -8 or -9)
        draw_icon(warning.kind, x, y + icon_offset, alpha, state.scale)

        local detail = warning.kind == "he" and warning.damage > 0 and string.format("-%d HP", warning.damage)
            or nil
        if detail then
            renderer_text(x, y + 12, 255, 255, 255, alpha, "cb", 0, detail)
        end
        renderer_text(
            x,
            y + (detail and 24 or 12),
            255,
            255,
            255,
            alpha,
            "cb",
            0,
            string.format("%.1f s", ceil(warning.remaining * 10) / 10)
        )
        if offscreen then
            draw_arrow(x, y, width, height, 43, 255, 255, 255, alpha)
        end
    end

    local function draw_warnings(warnings, ex, ey, ez)
        local nearby = {}
        for _, warning in ipairs(warnings) do
            local dx, dy, dz = warning.x - ex, warning.y - ey, warning.z - ez
            warning.distance = warning.fire_distance or sqrt(dx * dx + dy * dy + dz * dz)
            if warning.distance <= RANGE then
                nearby[#nearby + 1] = warning
            end
        end
        table.sort(nearby, function(a, b)
            return a.distance < b.distance
        end)
        local selected = {}
        for index = 1, min(#nearby, MAX_WARNINGS) do
            selected[nearby[index].state] = true
        end
        local now = globals_realtime()
        for _, warning in ipairs(warnings) do
            local state = warning.state
            local target = selected[state] and 1 or 0
            local transition = state.warning_transition
            if not transition then
                transition = { from = 0, value = 0, target = target, started = now }
                state.warning_transition = transition
            end
            transition.value = transition.from
                + (transition.target - transition.from)
                    * ease_out_cubic((now - transition.started) / 0.22)
            if transition.target ~= target then
                transition.from, transition.target, transition.started = transition.value, target, now
            end

            state.warning_visibility = transition.value * smoothstep((RANGE - warning.distance) / 24)

            if warning.kind ~= "fire" and warning.active ~= false then
                local enemy_danger = false
                local has_damage = (warning.damage or 0) > 0
                if warning.kind == "he" then
                    for _, entry in ipairs(warning.damage_entries or {}) do
                        if (entry.damage or 0) > 0 then
                            has_damage = true
                            enemy_danger = enemy_danger or entry.enemy == true
                        end
                    end
                elseif warning.kind == "molotov" then
                    enemy_danger = warning.state.enemy_danger == true
                end
                local r, g, b, alpha = trajectory_color()
                if enemy_danger then
                    r, g, b = GREEN[1], GREEN[2], GREEN[3]
                elseif has_damage then
                    r, g, b = YELLOW[1], YELLOW[2], YELLOW[3]
                end
                draw_trajectory(warning, r, g, b, alpha)
            end
            draw_damage_labels(warning)
        end
        local width, height = client.screen_size()
        local fire_reach = cvar_float("inferno_max_range", 150)
        for index = #nearby, 1, -1 do
            if (nearby[index].state.warning_visibility or 0) > 0 then
                draw_warning(nearby[index], width, height, fire_reach)
            end
        end
    end

    local function on_paint()
        sync_builtins(ui_get(enabled))
        if not ui_get(enabled) then
            reset()
            return
        end
        local player = entity.get_local_player()
        local ex, ey, ez = client_eye_position()
        if not player or not entity.is_alive(player) or not ex then
            return
        end
        check_trace(player)
        trace_ignore = player
        update_damage_targets(player)
        local warnings = {}
        local friendly_fire = cvar_float("mp_friendlyfire", 0) ~= 0
        add_projectiles(warnings, player, friendly_fire)
        add_fires(warnings, player, friendly_fire)
        draw_warnings(warnings, ex, ey, ez)
    end

    local function on_detonate(event)
        local id = tonumber(event.entityid)
        if id and id > 2047 then
            id = band(id, 0x7FF)
        end
        if id and tracked[id] then
            finish_projectile(id)
            return
        end

        if not event.x or not event.y or not event.z then
            return
        end
        for index, state in pairs(tracked) do
            local warning = state.last_warning
            local kind = event.name == "hegrenade_detonate" and "he" or "molotov"
            if warning and warning.kind == kind then
                local dx, dy, dz = warning.x - event.x, warning.y - event.y, warning.z - event.z
                local distance2 = dx * dx + dy * dy + dz * dz
                local current = state.current_position
                if current then
                    dx, dy, dz = current[1] - event.x, current[2] - event.y, current[3] - event.z
                    distance2 = min(distance2, dx * dx + dy * dy + dz * dz)
                end
                if distance2 < 128 * 128 then
                    finish_projectile(index)
                end
            end
        end
    end
    for _, event_name in ipairs({ "hegrenade_detonate", "molotov_detonate", "inferno_startburn" }) do
        local name = event_name
        client.set_event_callback(name, function(event)
            on_detonate({ name = name, entityid = event.entityid, x = event.x, y = event.y, z = event.z })
        end)
    end
    client.set_event_callback("paint", guarded(on_paint))
    client.set_event_callback("level_init", reset)
    client.set_event_callback("cs_game_disconnected", reset)
    client.set_event_callback("shutdown", function()
        sync_builtins(false)
    end)
end)()
